    <script>
      function formatDate(dateStr) {
        if (!dateStr) return "";
        const date = new Date(dateStr);
        return date.toLocaleDateString("en-US", {
          year: "numeric",
          month: "long",
          day: "numeric",
        });
      }

      function createTagObjects(tags) {
        if (!tags.trim()) return "";
        return tags
          .split(",")
          .map((t) => t.trim())
          .filter(Boolean)
          .map(
            (tag) =>
              `{'name': '${tag}', 'slug': '${tag
                .toLowerCase()
                .replace(/\s+/g, "-")}'}`
          )
          .join(",");
      }

      function generateMarkdown() {
        const slug = document.getElementById("slug").value;
        const title = document.getElementById("title").value;
        const description = document.getElementById("description").value;
        const content = document.getElementById("content").value;
        const publishedDate = document.getElementById("publishedDate").value;
        const modifiedDate = document.getElementById("modifiedDate").value;
        const image = document.getElementById("image").value;
        const keywords = document.getElementById("keywords").value;
        const tags = document.getElementById("tags").value;
        const urlPath = document.getElementById("urlPath").value;
        const urlSlug = document.getElementById("urlSlug").value;

        const formattedPublished = formatDate(publishedDate);
        const formattedModified = formatDate(modifiedDate);
        const tagObjects = createTagObjects(tags);

        let markdown = `<span class="markdown-syntax">---</span>
<span class="markdown-meta">slug:</span> <span class="markdown-content">${slug}</span>
<span class="markdown-meta">title:</span> <span class="markdown-title">"${title}"</span>
<span class="markdown-meta">description:</span> <span class="markdown-content">"${description}"</span>
<span class="markdown-meta">published_date:</span> <span class="markdown-content">${publishedDate}</span>
<span class="markdown-meta">formatted_date:</span> <span class="markdown-content">"${formattedPublished}"</span>
<span class="markdown-meta">modified_date:</span> <span class="markdown-content">${modifiedDate}</span>
<span class="markdown-meta">image:</span> <span class="markdown-content">"${image}"</span>
<span class="markdown-meta">keywords:</span> <span class="markdown-content">"${keywords}"</span>
<span class="markdown-meta">tags:</span> <span class="markdown-content">[${tagObjects}]</span>
<span class="markdown-meta">url:</span> <span class="markdown-content">"${urlPath}"</span>
<span class="markdown-meta">url_slug:</span> <span class="markdown-content">"${urlSlug}"</span>
<span class="markdown-syntax">---</span>

<span class="markdown-title"># ${title}</span>

${content}`;

        if (!title && !content && !slug) {
          markdown = `<span class="markdown-meta"># 📝 Start filling out the form to see your markdown preview here!

## What you'll see:
- YAML front matter with all your metadata
- Formatted title and content
- Real-time updates as you type

## Features:
- ✅ Auto-generated tag objects
- ✅ Formatted dates
- ✅ URL encoding ready
- ✅ Markdown syntax highlighting</span>`;
        }

        return markdown;
      }

      function updatePreview() {
        const previewElement = document.getElementById("markdownPreview");
        previewElement.innerHTML = generateMarkdown();
      }

      // Add event listeners to all form inputs
      const inputs = document.querySelectorAll("input, textarea");
      inputs.forEach((input) => {
        input.addEventListener("input", updatePreview);
      });

      // Initial preview update
      updatePreview();

      // Set default dates
      const now = new Date();
      const localDateTime = now.toISOString().slice(0, 16);
      document.getElementById("publishedDate").value = localDateTime;
      document.getElementById("modifiedDate").value = localDateTime;
      updatePreview();

      document
        .getElementById("blogForm")
        .addEventListener("submit", function (e) {
          e.preventDefault();

          const slug = encodeURIComponent(
            document.getElementById("slug").value
          );
          const title = encodeURIComponent(
            document.getElementById("title").value
          );
          const description = encodeURIComponent(
            document.getElementById("description").value
          );
          const content = encodeURIComponent(
            document.getElementById("content").value
          );

          const publishedDate = document.getElementById("publishedDate").value;
          const formattedDate = encodeURIComponent(formatDate(publishedDate));

          const modifiedDate = document.getElementById("modifiedDate").value;
          const image = encodeURIComponent(
            document.getElementById("image").value
          );

          const keywords = encodeURIComponent(
            document.getElementById("keywords").value
          );
          const tagsRaw = document.getElementById("tags").value;
          const tags = encodeURIComponent(createTagObjects(tagsRaw));

          const url = encodeURIComponent(
            document.getElementById("urlPath").value
          );
          const finalSlug = encodeURIComponent(
            document.getElementById("urlSlug").value
          );

          // Create URL with query parameters instead of path parameters
          const params = new URLSearchParams({
            slug: document.getElementById("slug").value,
            title: document.getElementById("title").value,
            description: document.getElementById("description").value,
            content: document.getElementById("content").value,
            publishedDate: publishedDate,
            formattedDate: formatDate(publishedDate),
            modifiedDate: modifiedDate,
            image: document.getElementById("image").value,
            keywords: document.getElementById("keywords").value,
            tags: createTagObjects(document.getElementById("tags").value),
            urlPath: document.getElementById("urlPath").value,
            urlSlug: document.getElementById("urlSlug").value,
          });

          const finalURL = `/upload?${params.toString()}`;

          // Redirect
          window.location.href = finalURL;
        });
